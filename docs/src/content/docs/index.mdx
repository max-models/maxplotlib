```python
import numpy as np
from maxplotlib import Canvas

x = np.linspace(0, 2 * np.pi, 200)

canvas, ax = Canvas.subplots()
ax.plot(x, np.sin(x))

canvas.show()                      # Matplotlib, the default
canvas.show(backend="plotly")      # interactive
canvas.show(backend="plotext")     # in the terminal
canvas.savefig("sine.pdf", backend="tikzfigure")  # TikZ, compiled with LaTeX
```
